@celery_app.task(name="tasks.fatorial")
def fatorial(n):
    if n < 0:
        raise ValueError("O número deve ser maior ou igual a zero")

    resultado = 1
    for i in range(2, n + 1):
        resultado *= i

    return resultado

# Envio da tarefa para execução em segundo plano
tarefa = fatorial.delay(5)

// Busca por nome e filtro por categoria
function applyFilters() {
  const search = searchInput.value
    .trim()
    .toLocaleLowerCase("pt-BR");

  productCards.forEach(column => {
    const product = column.querySelector(".product-card");
    const name = product.dataset.name.toLocaleLowerCase("pt-BR");
    const categories = product.dataset.categories.split(" ");

    const matchesCategory = activeCategory === "all" ||
      categories.includes(activeCategory);
    const matchesSearch = name.includes(search);

    column.hidden = !(matchesCategory && matchesSearch);
  });
}

// Exemplo de cálculo e atualização da interface
function calcularIMC() {
  const peso = Number(document.getElementById("peso").value);
  const altura = Number(document.getElementById("altura").value);

  if (peso <= 0 || altura <= 0) {
    resultado.textContent = "Informe valores válidos.";
    return;
  }

  const imc = peso / (altura * altura);
  resultado.textContent = `Seu IMC é ${imc.toFixed(2)}`;
}
